Find Your Best Match

The Personalised Country Wellbeing Score (PCWS) is an interactive country recommendation tool based on OECD wellbeing data. Rather than identifying one country as universally “best”, the model evaluates countries according to a user’s demographic profile and the aspects of wellbeing that matter most to that individual.

The user provides:

  • education level
  • age group
  • desired annual income
  • preferred weekly working hours
  • the relative importance of happines, financial satisfication, leisure satisfaction, employment security, income and working hours.

These inputs are combined with OECD indicators to produce a personalised score for each country and a ranked recommendation.

Interpretation and limitations

The Personalised Country Wellbeing Score should be interpreted as a recommendation index, not as an individual causal prediction of future happiness.

The current dataset contains aggregated OECD observations rather than individual-level records. In addition, the wellbeing percentages and labour-market indicators are drawn from related but not identical demographic groupings.

The model therefore answers a question closer to:

Which countries have observed wellbeing and labour-market profiles that most closely match this user’s characteristics and stated preferences?

It does not establish that moving to the highest-ranked country would cause that individual to become happier.

The user-defined weighting system is also intentionally subjective. This is a feature of the model rather than an error: the purpose is to make the value judgements behind the recommendation visible and adjustable instead of embedding fixed assumptions about what everyone should value.

#Income compatability
income_match <- function(country_income, desired_income) {

  difference <- abs(country_income - desired_income)

  # Exponential, No difference → score of 100. Bigger difference → rapidly smaller score.
  #"Small differences are meaningful, but once two things are already very different, additional difference doesn't matter very much."
  # differences near zero matter a lot; differences that are already large matter progressively less.
  #smaller denominator → score falls faster → more sensitive to differences
  #larger denominator → score falls slower → less sensitive to differences
  #So the denominator defines the scale of difference that you consider substantial.

  score <- 100 * exp(
    -difference / desired_income
  )

  return(score)
}

#Numbers of working hours compatability
hours_match <- function(country_hours, desired_hours) {

  difference <- abs(country_hours - desired_hours)

  #smaller 10 → score falls faster → more sensitive to differences
  #larger 10 → score falls slower → less sensitive to differences
  #when difference = 10, your score has fallen to exactly 100e−1≈36.8.

  score <- 100 * exp(
    -difference / 10
  )

  return(score)
}

server <- function(input, output, session) {

  #Starting the Server

  #Update the age choices so only age groups available for the chosen
  #education level can be selected
  observeEvent(input$education, {

    available_ages <- df %>%
      filter(
        education == input$education,
        age_group != "All ages"
      ) %>%
      distinct(age_group) %>%
      pull(age_group)

    updateSelectInput(
      session,
      "age_group",
      choices = available_ages,
      selected = if (input$age_group %in% available_ages) {
        input$age_group
      } else {
        available_ages[1]
      }
    )
  })

  #Getting the user data, to get the correct reference group
  results <- reactive({

    #The wellbeing indicators are available by education at "All ages".
    #Therefore these are retrieved separately from the age-specific
    #labour market indicators.
    wellbeing_data <- df %>%
      filter(
        education == input$education,
        age_group == "All ages"
      ) %>%
      select(
        iso3,
        country,
        pct_happy,
        pct_satisfied_finances,
        pct_satisfied_leisure
      )

    #Getting age-specific work and economic data for the user's education group
    work_data <- df %>%
      filter(
        education == input$education,
        age_group == input$age_group
      ) %>%
      select(
        iso3,
        country,
        earnings_gross_yearly,
        hours_worked,
        pct_permanent_contract
      )

    #Combining the two OECD reference groups by country
    data <- wellbeing_data %>%
      inner_join(
        work_data,
        by = c("iso3", "country")
      )

    #Removing countries where one or more indicators required
    #for the score are unavailable
    data <- data %>%
      filter(
        !is.na(pct_happy),
        !is.na(pct_satisfied_finances),
        !is.na(pct_satisfied_leisure),
        !is.na(earnings_gross_yearly),
        !is.na(hours_worked),
        !is.na(pct_permanent_contract)
      )

    #Check that we have data
    validate(
      need(
        nrow(data) > 0,
        "No complete data is available for this education and age combination."
      )
    )

    #Getting the user input importance "weights"
    weights <- c(
      happiness = input$weight_happiness,
      finance = input$weight_finance,
      leisure = input$weight_leisure,
      contract = input$weight_contract,
      income = input$weight_income,
      hours = input$weight_hours
    )

    #Normalising the weights so that the sum is always 1 (100%)
    if (sum(weights) == 0) {
      weights <- rep(1, length(weights))
      names(weights) <- c(
        "happiness", "finance", "leisure",
        "contract", "income", "hours"
      )
    }

    weights <- weights / sum(weights)

    # Making the scores for the indicators, all except income and time from raw data
    # + adding the contribution facotrs from the user
    # + adding the total calculated country wellbeing score for the user
    data <- data %>%
      mutate(

        happiness_score =
          pct_happy,

        contribution_happiness =
          weights["happiness"] *
          happiness_score,

        finance_score =
          pct_satisfied_finances,

        contribution_finance =
          weights["finance"] *
          finance_score,

        leisure_score =
          pct_satisfied_leisure,

        contribution_leisure =
          weights["leisure"] *
          leisure_score,

        contract_score =
          pct_permanent_contract,

        contribution_contract =
          weights["contract"] *
          contract_score,

        income_match_score =
          income_match(
            earnings_gross_yearly,
            input$income
          ),

        contribution_income =
          weights["income"] *
          income_match_score,

        hours_match_score =
          hours_match(
            hours_worked,
            input$hours
          ),

        contribution_hours =
          weights["hours"] *
          hours_match_score,

        personalised_score =
          contribution_happiness +
          contribution_finance +
          contribution_leisure +
          contribution_contract +
          contribution_income +
          contribution_hours
      )

    #Reformatting datatable, one country per row
    #After combining the education and age reference groups there is already
    #one usable row per country, so the columns are renamed here for the outputs.
    data <- data %>%
      arrange(desc(personalised_score)) %>%
      mutate(
        rank = row_number()
      ) %>%
      transmute(
        rank = rank,
        country = country,
        iso3 = iso3,
        score = personalised_score,
        happiness = happiness_score,
        finance = finance_score,
        leisure = leisure_score,
        contract = contract_score,
        income = earnings_gross_yearly,
        hours = hours_worked,
        income_match = income_match_score,
        hours_match = hours_match_score,
        contribution_happiness = contribution_happiness,
        contribution_finance = contribution_finance,
        contribution_leisure = contribution_leisure,
        contribution_contract = contribution_contract,
        contribution_income = contribution_income,
        contribution_hours = contribution_hours
      )

    return(data)
  })

  # The results
  output$summary <- renderUI({

    best <- results() %>%
      slice(1)

    tags$div(
      class = "wellbeing-result",
      tags$h2(
        paste(
          "Your recommended country is",
          best$country
        )
      ),
      tags$p(
        paste0(
          "Personalised Country Wellbeing Score: ",
          round(best$score, 1),
          " / 100"
        )
      )
    )
  })

  # Why was that country selected for you?
  output$why_country <- renderUI({

    best <- results() %>%
      slice(1)

    contributions <- data.frame(
      factor = c(
        "Happiness",
        "Financial satisfaction",
        "Leisure satisfaction",
        "Employment security",
        "Income compatibility",
        "Working-hours compatibility"
      ),
      contribution = c(
        best$contribution_happiness,
        best$contribution_finance,
        best$contribution_leisure,
        best$contribution_contract,
        best$contribution_income,
        best$contribution_hours
      )
    ) %>%
      arrange(desc(contribution))

    top_factors <- contributions %>%
      slice_head(n = 3)

    tags$div(
      tags$p(
        paste0(
          best$country,
          " ranks first for your profile."
        )
      ),
      tags$p(
        "Its three strongest contributors are:"
      ),
      tags$ol(
        tags$li(top_factors$factor[1]),
        tags$li(top_factors$factor[2]),
        tags$li(top_factors$factor[3])
      ),
      tags$p(
        paste0(
          round(best$happiness, 1),
          "% of the relevant education group report being happy."
        )
      ),
      tags$p(
        paste0(
          "Income compatibility: ",
          round(best$income_match, 1),
          "/100. Working-hours compatibility: ",
          round(best$hours_match, 1),
          "/100."
        )
      )
    )
  })

  # Weight contribution charts for the different indicators
  output$contribution_plot <- renderPlot({

    best <- results() %>%
      slice(1)

    contribution_data <- data.frame(
      factor = c(
        "Happiness",
        "Financial satisfaction",
        "Leisure",
        "Employment security",
        "Income match",
        "Working hours"
      ),
      contribution = c(
        best$contribution_happiness,
        best$contribution_finance,
        best$contribution_leisure,
        best$contribution_contract,
        best$contribution_income,
        best$contribution_hours
      )
    )

    ggplot(
      contribution_data,
      aes(
        x = reorder(factor, contribution),
        y = contribution
      )
    ) +
      geom_col() +
      coord_flip() +
      labs(
        title = paste(
          "Why",
          best$country,
          "ranks first"
        ),
        x = NULL,
        y = "Contribution to score"
      ) +
      theme_minimal()
  })

  #Your top 10 countries table
  output$country_table <- renderTable({

    results() %>%
      slice_head(n = 10) %>%
      transmute(
        Rank = rank,
        Country = country,
        `Wellbeing Score` =
          round(score, 1),
        `Happy %` =
          round(happiness, 1),
        `Financially Satisfied %` =
          round(finance, 1),
        `Leisure Satisfied %` =
          round(leisure, 1),
        `Permanent Contract %` =
          round(contract, 1),
        `Income Match` =
          round(income_match, 1),
        `Hours Match` =
          round(hours_match, 1)
      )

  }, striped = TRUE, bordered = TRUE, spacing = "s")

  # Your top 10 countries plot
  output$country_plot <- renderPlot({

    plot_data <- results() %>%
      slice_head(n = 10)

    ggplot(
      plot_data,
      aes(
        x = reorder(country, score),
        y = score
      )
    ) +
      geom_col() +
      coord_flip() +
      labs(
        title = "Your Top 10 Country Matches",
        x = NULL,
        y = "Personalised Wellbeing Score"
      ) +
      theme_minimal()
  })
}

shinyApp(
  ui = ui,
  server = server
)

Role in the model

The variables beginning with pct_ are already expressed as percentages in the OECD data. For example, a value of 82 in pct_happy means that 82% of the relevant group reports being happy.

The percentage variables stays on their original 0–100 scale.

3. Measuring income and working-hours compatibility

Income and working hours are measured in numerical values and not percentages. In this format it cannot be inserted into the same weighted score as the wellbeing indicators. Instead, the model converts the difference between observed earnings and the user’s desired earnings into a 0–100 compatibility score. The same is applied to working hours.

An exact match between observed earnings and the user’s desired income receives a compatibility score of 100. As the difference increases, the score decreases.

This means the model asks:

How closely do observed earnings in this country match the user’s preferred income level?

rather than:

Which country simply has the highest earnings?

The PCWS is designed as a personalised compatibility model. A higher income is therefore not automatically treated as better. The exponential function gives relatively greater importance to differences close to the user’s target and progressively less importance to additional differences once two values are already far apart.

For the “Working hours” the denominator 10 controls how sensitive the score is to differences in working hours. It is a modelling assumption and should be interpreted as part of the model specification.

#Making the functions for compatability
#Transforming the values so that they match the other indicators % style on a scale of 0-100

#Income compatability
income_match <- function(country_income, desired_income) {

  difference <- abs(country_income - desired_income)

  # Exponential, No difference → score of 100. Bigger difference → rapidly smaller score.
  #"Small differences are meaningful, but once two things are already very different, additional difference doesn't matter very much."
  # differences near zero matter a lot; differences that are already large matter progressively less.
  #smaller denominator → score falls faster → more sensitive to differences
  #larger denominator → score falls slower → less sensitive to differences
  #So the denominator defines the scale of difference that you consider substantial.

  score <- 100 * exp(
    -difference / desired_income
  )

  return(score)
}

#Numbers of working hours compatability
hours_match <- function(country_hours, desired_hours) {

  difference <- abs(country_hours - desired_hours)

  #smaller 10 → score falls faster → more sensitive to differences
  #larger 10 → score falls slower → less sensitive to differences
  #when difference = 10, your score has fallen to exactly 100e−1≈36.8.

  score <- 100 * exp(
    -difference / 10
  )

  return(score)
}

8. Combining the two OECD reference groups

A central feature of the current dataset is that the wellbeing indicators and labour-market indicators are not available at exactly the same demographic level.

The wellbeing percentages are available for the selected education group across "All ages", while income, working hours and permanent-contract information are available for the selected education and age group.

The model therefore retrieves the two sets of indicators separately and combines them by country.

After the join, each country has one profile containing:

  • happiness for the selected education group
  • financial satisfaction for the selected education group
  • leisure satisfaction for the selected education group
  • age-specific earnings for the selected education group
  • age-specific working hours
  • age-specific permanent-contract prevalence

This also introduces an important methodological limitation: the final score combines indicators from related but not identical demographic aggregation levels.

9. Handling missing country observations

Some countries may still be missing one or more of the indicators required to calculate the complete score. To avoid NaN values and prevening the model from inventing or imputing observations that are not present in the OECD data only countries with complete information for all six score components remain in the ranking.

Therefore the number of countries included in the ranking may vary between user profiles.

The six slider values are first collected into a vector and then nornmalised so they always add up to 100%.

11. Calculating the Personalised Country Wellbeing Score

Once the country indicators and user weights are available, each component is multiplied by its corresponding weight.

The final score can be written as:

[ PCWS_c = w_HH_c + w_FF_c + w_LL_c + w_EE_c + w_II_c + w_WW_c ]

where:

  • (H) = happiness percentage
  • (F) = financial-satisfaction percentage
  • (L) = leisure-satisfaction percentage
  • (E) = permanent-contract percentage
  • (I) = income compatibility
  • (W) = working-hours compatibility
  • each (w) represents the user’s normalised importance weight.

All six component measures use a 0–100 scale, and the weights sum to 1. The final PCWS therefore also lies on a 0–100 scale.

The score should be interpreted as a weighted compatibility index. It represents how strongly a country’s observed profile matches the user’s demographic characteristics and stated preferences.

# Making the scores for the indicators, all except income and time from raw data
# + adding the contribution facotrs from the user
# + adding the total calculated country wellbeing score for the user
data <- data %>%
  mutate(

    happiness_score =
      pct_happy,

    contribution_happiness =
      weights["happiness"] *
      happiness_score,


    finance_score =
      pct_satisfied_finances,

    contribution_finance =
      weights["finance"] *
      finance_score,


    leisure_score =
      pct_satisfied_leisure,

    contribution_leisure =
      weights["leisure"] *
      leisure_score,


    contract_score =
      pct_permanent_contract,

    contribution_contract =
      weights["contract"] *
      contract_score,


    income_match_score =
      income_match(
        earnings_gross_yearly,
        input$income
      ),

    contribution_income =
      weights["income"] *
      income_match_score,


    hours_match_score =
      hours_match(
        hours_worked,
        input$hours
      ),

    contribution_hours =
      weights["hours"] *
      hours_match_score,


    personalised_score =
      contribution_happiness +
      contribution_finance +
      contribution_leisure +
      contribution_contract +
      contribution_income +
      contribution_hours
  )
