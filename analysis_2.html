All the necessary information are located in the data_table$Indicator column. We need to fetch some more information from there.

Pivot wider and reformat

So education lvl, age groups, different percentages, working hours, etc. become separate columns.

# VAR is a structured code: "<numerator>X<denominator>" for shares, or a plain
# code for levels. The numerator encodes education, measure and age band, e.g.
#   PHD_TOC_PERM_35_44XPHD_TOC_35_44   MAS_EARNY_25_34   TOTWB_HAPPY_GESMTXTOTWB
measure_labels <- c(
  "_TOC_PERM"        = "pct_permanent_contract",
  "_HW"              = "hours_worked",
  "_EARNY"           = "earnings_gross_yearly",
  "_HAPPY_GESMT"     = "pct_happy",
  "_FINS_MEDHIGH"    = "pct_satisfied_finances",
  "_TIMEUSE_MEDHIGH" = "pct_satisfied_leisure"
)

# Observed values first, then caveated, then imputed. Q is always a missing value.
status_priority <- c("A", "B", "D", "U", "V", "I", "Q")

data_long <- data_table |>
  dplyr::mutate(
    num_code  = str_remove(VAR, "X.*$"),
    core_code = num_code |>
      str_remove("^(TOT|MAS|PHD)(WB)?") |>
      str_remove("_\\d{2}_\\d{2}$"),
    education = factor(str_extract(num_code, "^(TOT|MAS|PHD)"),
                       levels = c("TOT", "MAS", "PHD"),
                       labels = c("Total population", "Master's", "Doctorate")),
    age_group = num_code |>
      str_extract("\\d{2}_\\d{2}") |>
      str_replace("_", "-") |>
      replace_na("All ages") |>
      factor(levels = c("All ages", "25-34", "35-44", "45-54", "55-64")),
    measure   = unname(measure_labels[core_code])
  )

# Fail loud if a VAR code ever stops matching the pattern above.
stopifnot(!anyNA(data_long$education), !anyNA(data_long$age_group),
          !anyNA(data_long$measure))

# 154 country x VAR pairs appear 2-3 times under different observation statuses.
# Keep the best-status rows, then average whatever is still tied.
data_long <- data_long |>
  dplyr::mutate(status_rank = match(OBS_STATUS_1, status_priority)) |>
  slice_min(status_rank, by = c(REF_AREA, VAR), with_ties = TRUE) |>
  summarise(
    value    = if (all(is.na(OBS_VALUE))) NA_real_ else mean(OBS_VALUE, na.rm = TRUE),
    n_source = n(),
    .by = c(REF_AREA, `Reference area`, VAR, Indicator,
            education, age_group, measure, `Observation status`)
  ) |>
  rename(iso3 = REF_AREA, country = `Reference area`, status = `Observation status`) |>
  arrange(country, education, age_group, measure)

# One row per country x education x age band, one column per measure.
data_wide <- data_long |>
  select(iso3, country, education, age_group, measure, value) |>
  pivot_wider(names_from = measure, values_from = value) |>
  arrange(country, education, age_group)

# Correct the country name for Turkey since was encoded as latin1, and now back to UTF-8
data_wide <- data_wide |> dplyr::mutate(country = if_else(iso3 == "TUR", "Türkiye", country))

# The original Indicator text cannot be a column in the wide table (each row
# spans up to 6 indicators), so keep it as a lookup for axis labels / tooltips.
indicator_dict <- data_long |>
  distinct(education, age_group, measure, VAR, Indicator)

Correct the income values of Croatia:

# Croatia reports earnings in kuna while every other country is on the survey's
# harmonised unit. Mean gross yearly earnings across all EARNY variables:
# HRV 322 743, against 122 411 for Luxembourg, the next highest -- a 2.6x gap
# with no counterpart in any other indicator. Dividing by the irrevocable
# HRK/EUR rate fixed at Croatia's euro entry (1 Jan 2023) lands Croatia at
# 42 835, between Estonia and Poland, which matches its position elsewhere.
HRK_PER_EUR <- 7.53450

# Before/after, for the write-up. Run this *before* the mutate below.
data_wide |>
  filter(iso3 == "HRV", !is.na(earnings_gross_yearly)) |>
  transmute(education, age_group,
            as_reported = round(earnings_gross_yearly),
            in_euro     = round(earnings_gross_yearly / HRK_PER_EUR))
